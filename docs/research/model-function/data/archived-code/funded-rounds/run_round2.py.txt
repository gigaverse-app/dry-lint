def functions(folder,b):
 manifest=json.loads((folder/'source-manifest.json').read_text());found=collections.defaultdict(list);missing=[]

  cp=FOLLOW/'cache'/(sha((p['path']+'\n'+p['sha256']+'\n'+ex.version+'\n'+mex.version).encode())+'.json.gz')
  if not cp.exists():missing.append(p['path']);continue
  cap=json.loads(gzip.decompress(cp.read_bytes()))
  for r in cap['functions']:
   f=obj(r);found[f.qualname].append(f)
 docs=collections.defaultdict(list)
 for d in b.documents:docs[d.id].append(d)
 fs={};counts=collections.Counter()
 for fid,ds in docs.items():
  if len(ds)!=1 or len(found.get(fid,[]))!=1:counts['missing_or_ambiguous_native']+=1;continue
  f=found[fid][0]
  if sha(f.code.encode())!=ds[0].source_sha256:counts['native_body_mismatch']+=1;continue
  if f.line<1 or f.end_line<f.line:counts['invalid_native_span']+=1;continue
  f.tokens=ds[0].tokens();fs[fid]=f
 return fs,dict(counts),missing
