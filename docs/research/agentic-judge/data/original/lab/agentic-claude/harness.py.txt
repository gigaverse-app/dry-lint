def canonical(obj):
    return json.dumps(obj, sort_keys=True, separators=(',', ':'), ensure_ascii=False).encode()


def sha(obj):
    return hashlib.sha256(obj if isinstance(obj, bytes) else canonical(obj)).hexdigest()


def load_json(path):
    return json.loads(Path(path).read_text())


def lines(path):
    return [json.loads(s) for s in Path(path).read_text().splitlines() if s.strip()]


def save(path, obj):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + '.tmp')
    with temp.open('wb') as f:
        f.write(canonical(obj) + b'\n')
        f.flush()
        os.fsync(f.fileno())
    os.replace(temp, path)


def schema(name, description, properties, required):
    return {'name': name, 'description': description,
            'input_schema': {'type': 'object', 'properties': properties,
                             'required': required, 'additionalProperties': False}}

def initial_request(item):
    coords = [{'evidence_id': f'initial:{p["side"]}', 'path': p['path'],
               'start_line': p['line'], 'end_line': p['end_line']} for p in item['source']]
    return {'model': MODEL, 'max_tokens': LIMITS['output_tokens_per_round'],
            'system': SYSTEM + '\nInitial body coordinates: ' + canonical(coords).decode(),
            'messages': [{'role': 'user', 'content': item['rendered_body_only_prompt']}],
            'tools': TOOLS, 'tool_choice': {'type': 'auto'}}


def request_identity(item):
    return {'schema': 1, 'version': VERSION, 'source_sha256': item['source_identity_sha256'],
            'candidate_sha256': item['candidate_sha256'], 'request_sha256': sha(initial_request(item)),
            'tool_schemas_sha256': sha(TOOLS), 'model': MODEL, 'limits': LIMITS,
            'harness_sha256': sha(Path(__file__).read_bytes()),
            'cache_policy': 'fresh execution; verified exact experimental cache only, never production store',
            'pricing': {'input_per_million': 2.0, 'output_per_million': 10.0},
            'no_retries': True, 'provider_prompt_cache_requested': False}


def estimated_cost(request):
    return round((len(canonical(request)) / 2.8 * 2 + 512 * 10) / 1e6, 8)


def reservation(request):
    # UTF-8 bytes bound ordinary BPE text tokens; fixed 4096 allowance for framing.
    # Root must validate pricing and this documented reservation policy before executing.
    return round((len(canonical(request)) * 2 + 4096 * 2 + request['max_tokens'] * 10) / 1e6, 8)


def valid_path(path):
    if not isinstance(path, str) or not path or len(path) > 400:
        return False
    p = PurePosixPath(path)
    bad_parts = {'.git', '.aws', '.ssh', '.codex', '.agents', '.lab-results', 'enterprise',
                 'node_modules', 'reports', 'research', 'credentials', '__pycache__'}
    return (not p.is_absolute() and '..' not in p.parts and not path.startswith('-') and
            not any(c in path for c in '\\:\x00\n\r') and all(x not in ('', '.') for x in path.split('/')) and
            not (set(x.lower() for x in p.parts) & bad_parts) and
            p.suffix.lower() in {'.py', '.ts', '.tsx', '.js', '.jsx'} and
            not re.search(r'(^|/)(?:\.env|secret|credential|private[-_]key)(?:[._/]|$)', path, re.I))


SECRET = re.compile(r'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----|'
                    r'\b(?:sk-(?:ant-|proj-)?[A-Za-z0-9_-]{20,}|AKIA[0-9A-Z]{16}|gh[pousr]_[A-Za-z0-9]{20,})\b|'
                    r'(?i:(?:api[_-]?key|access[_-]?token|password|client[_-]?secret)\s*[:=]\s*[\'\"])[^\'\"\n\s]{12,}')


def source_safe(text):
    # Fail closed on probable credential literals; never echo the rejected content.
    if SECRET.search(text):
        raise ValueError('source withheld: possible credential literal')


class SourceTools:
    def __init__(self, config, candidate, allowed_paths=None):
        self.config = copy.deepcopy(config)
        self.repo = config['repository']
        self.pin = config['revision']
        if not re.fullmatch('[0-9a-f]{40}', self.pin):
            raise ValueError('full commit SHA required')
        self.git_dir = str(Path(config['git']).resolve())
        self.candidate = candidate
        self.returned_chars = self.returned_bytes = self.tool_calls = 0
        self.catalog = {}
        self._files = {}
        if allowed_paths is None:
            proof = config['background_proof']
            raw = Path(proof['path']).read_bytes()
            if sha(raw) != proof['sha256']:
                raise ValueError('background hash mismatch')
            bg = json.loads(gzip.decompress(raw))
            if (bg['repository'], bg['revision']) != (self.repo, self.pin):
                raise ValueError('background pin mismatch')
            allowed_paths = {d['id'].split('::')[0] for d in bg['documents']}
        tree = self.git('ls-tree', '-r', '-z', self.pin).split(b'\0')
        regular = set()
        for entry in tree:
            if b'\t' not in entry:
                continue
            meta, path = entry.split(b'\t', 1)
            if meta.split()[0] not in (b'100644', b'100755'):
                continue  # no symlinks, submodules or tree entries
            p = path.decode('utf-8', 'strict')
            if p in allowed_paths and valid_path(p):
                regular.add(p)
        self.paths = sorted(regular)
        self.allowed = set(self.paths)
        if not all(candidate[s]['path'] in self.allowed for s in ('new', 'existing')):
            raise ValueError('endpoint outside frozen public source scope')
        for side in ('new', 'existing'):
            f = candidate[side]
            source_safe(f['code'])
            self.catalog[f'initial:{side}'] = {'path': f['path'], 'start_line': f['line'], 'end_line': f['end_line']}

    def git(self, *args):
        # No credential variables, shell, network, replace refs, hooks, pager or external diff.
        env = {'PATH': '/usr/bin:/bin', 'HOME': '/nonexistent', 'LC_ALL': 'C.UTF-8',
               'GIT_CONFIG_NOSYSTEM': '1', 'GIT_CONFIG_GLOBAL': '/dev/null',
               'GIT_NO_REPLACE_OBJECTS': '1', 'GIT_TERMINAL_PROMPT': '0', 'GIT_NO_LAZY_FETCH': '1'}
        cmd = ['/usr/bin/git', '--no-replace-objects', '-c', 'core.fsmonitor=false',
               '-c', 'core.hooksPath=/dev/null', '-C', self.git_dir, *args]
        result = subprocess.run(cmd, stdin=subprocess.DEVNULL, capture_output=True, env=env, timeout=10)
        if result.returncode:
            raise ValueError('pinned git object unavailable')
        return result.stdout

    def file(self, path):
        if not valid_path(path) or path not in self.allowed:
            raise ValueError('path outside allowed pinned source')
        if path not in self._files:
            size = int(self.git('cat-file', '-s', f'{self.pin}:{path}'))
            if size > LIMITS['source_file_bytes']:
                raise ValueError('source file exceeds byte limit')
            raw = self.git('show', f'{self.pin}:{path}')
            if b'\0' in raw:
                raise ValueError('binary source withheld')
            text = raw.decode('utf-8', 'strict')
            source_safe(text)
            self._files[path] = text
        return self._files[path]

    def region(self, path, start_line, end_line):
        if type(start_line) is not int or type(end_line) is not int or start_line < 1 or end_line < start_line:
            raise ValueError('invalid line range')
        text = self.file(path)
        source_lines = text.splitlines()
        if start_line > len(source_lines):
            raise ValueError('line outside file')
        end = min(end_line, len(source_lines), start_line + LIMITS['read_lines'] - 1)
        selected, chars = [], 0
        for line in source_lines[start_line - 1:end]:
            if chars + len(line) + 1 > LIMITS['read_chars']:
                break
            selected.append(line)
            chars += len(line) + 1
        if not selected:
            raise ValueError('single source line exceeds read cap')
        code = '\n'.join(selected)
        actual_end = start_line + len(selected) - 1
        fact = {'repository': self.repo, 'revision': self.pin, 'path': path,
                'start_line': start_line, 'end_line': actual_end, 'file_sha256': sha(text.encode()),
                'region_sha256': sha(code.encode()), 'code': code,
                'truncated': actual_end < end_line, 'requested_end_line': end_line}
        fact['evidence_id'] = 'source:' + sha(fact)[:24]
        return fact

    def symbols(self, path):
        text = self.file(path)
        if path.endswith('.py'):
            tree = ast.parse(text)
            found = []

            def walk(node, parents):
                for child in ast.iter_child_nodes(node):
                    if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                        names = parents + [child.name]
                        found.append(('.'.join(names), child.lineno, child.end_lineno))
                        walk(child, names)
                    else:
                        walk(child, parents)
            walk(tree, [])
            return found
        from dry_lint.typescript import TypeScriptExtractor
        return [(f'{f.owner}.{f.name}' if f.owner else f.name, f.line, f.end_line)
                for f in TypeScriptExtractor().functions(path, text)]

    def imports(self, path):
        text = self.file(path)
        found = []
        if path.endswith('.py'):
            for node in ast.walk(ast.parse(text)):
                if isinstance(node, (ast.Import, ast.ImportFrom)):
                    module = getattr(node, 'module', None)
                    level = getattr(node, 'level', 0)
                    targets = [module or ''] if isinstance(node, ast.ImportFrom) else [a.name for a in node.names]
                    resolved = []
                    for target in targets:
                        parts = list(PurePosixPath(path).parent.parts)
                        if level:
                            parts = parts[:len(parts) - level + 1]
                        else:
                            parts = []
                        base = '/'.join(parts + target.split('.')).strip('/')
                        resolved.extend(p for p in (base + '.py', base + '/__init__.py') if p in self.allowed)
                    found.append({'start_line': node.lineno, 'end_line': node.end_lineno,
                                  'text': ast.get_source_segment(text, node),
                                  'aliases': [{'name': a.name, 'asname': a.asname} for a in node.names],
                                  'resolved_paths': resolved, 'resolution': 'lexical local module only; binding not guaranteed'})
        else:
            pattern = r'(?m)^\s*(?:import|export)\s+[^;\n]*?(?:from\s*)?[\'\"]([^\'\"]+)[\'\"]'
            for m in re.finditer(pattern, text):
                module = m.group(1)
                resolved = []
                if module.startswith('.'):
                    parts = list(PurePosixPath(path).parent.parts)
                    for part in module.split('/'):
                        if part == '..':
                            if parts:
                                parts.pop()
                        elif part != '.':
                            parts.append(part)
                    base = '/'.join(parts)
                    resolved = [p for suffix in ('', '.ts', '.tsx', '.js', '/index.ts', '/index.tsx')
                                if (p := base + suffix) in self.allowed]
                found.append({'start_line': text[:m.start()].count('\n') + 1,
                              'end_line': text[:m.end()].count('\n') + 1, 'text': m.group(0).strip(),
                              'resolved_paths': resolved, 'resolution': 'relative lexical module only; package/tsconfig aliases unresolved'})
        return {'imports': found[:20], 'truncated': len(found) > 20,
                'file_sha256': sha(text.encode()), 'path': path, 'repository': self.repo, 'revision': self.pin}

    def references(self, symbol, prefix, limit):
        if prefix and (prefix.startswith(('/', '-')) or '..' in prefix.split('/') or
                       any(c in prefix for c in '\\:\x00\n\r')):
            raise ValueError('invalid prefix')
        term = symbol.split('.')[-1]
        roots = [str(PurePosixPath(self.candidate[s]['path']).parent) for s in ('new', 'existing')]
        candidates = [p for p in self.paths if p.startswith(prefix)]
        candidates.sort(key=lambda p: (not any(p.startswith(r + '/') for r in roots), p))
        chosen = candidates[:LIMITS['reference_scan_paths']]
        matches, unavailable = [], 0
        for path in chosen:
            try:
                text = self.file(path)
                if path.endswith('.py'):
                    tree = ast.parse(text)
                    refs = [(n.lineno, 'ast_call_lexical_binding_unknown') for n in ast.walk(tree)
                            if isinstance(n, ast.Call) and
                            ((isinstance(n.func, ast.Name) and n.func.id == term) or
                             (isinstance(n.func, ast.Attribute) and n.func.attr == term))]
                else:
                    refs = [(n, 'lexical_token_binding_unknown') for n, line in enumerate(text.splitlines(), 1)
                            if re.search(r'(?<![\w$])' + re.escape(term) + r'(?![\w$])', line)]
                for n, kind in sorted(refs):
                    fact = self.region(path, n, n)
                    fact['reference_kind'] = kind
                    matches.append(fact)
                    if len(matches) >= limit:
                        return {'matches': matches, 'truncated': True, 'scan_paths': len(chosen),
                                'available_paths': len(candidates), 'unavailable_files': unavailable,
                                'absence_is_not_proof': True}
            except (ValueError, UnicodeError, SyntaxError):
                unavailable += 1
        return {'matches': matches, 'truncated': len(chosen) < len(candidates), 'scan_paths': len(chosen),
                'available_paths': len(candidates), 'unavailable_files': unavailable, 'absence_is_not_proof': True}

    def call(self, name, args):
        self.tool_calls += 1
        if self.tool_calls > LIMITS['tool_calls']:
            return {'error': 'tool_call_cap', 'unknown': True}
        try:
            spec = next((s['input_schema'] for s in TOOLS if s['name'] == name), None)
            if spec is None or not isinstance(args, dict) or set(args) - set(spec['properties']) or not set(spec['required']) <= set(args):
                raise ValueError('unknown tool or malformed arguments')
            if 'symbol' in args and not re.fullmatch(r'[A-Za-z_$][\w$]*(?:\.[A-Za-z_$][\w$]*)*', args['symbol']):
                raise ValueError('invalid literal symbol')
            if name == 'list_paths':
                query, limit = args['query'], args.get('limit', 20)
                if not isinstance(query, str) or len(query) > 120 or type(limit) is not int or not 1 <= limit <= 30:
                    raise ValueError('invalid list bound')
                ps = [p for p in self.paths if query in p]
                result = {'paths': ps[:limit], 'truncated': len(ps) > limit, 'matches': len(ps),
                          'repository': self.repo, 'revision': self.pin}
            elif name == 'read_source':
                result = self.region(args['path'], args['start_line'], args['end_line'])
            elif name == 'read_symbol':
                hits = [x for x in self.symbols(args['path']) if x[0] == args['symbol']]
                if len(hits) != 1:
                    raise ValueError('symbol absent/ambiguous; use qualified name or line range')
                _, start, end = hits[0]
                result = self.region(args['path'], start, end)
            elif name == 'inspect_imports':
                result = self.imports(args['path'])
            else:
                limit = args.get('limit', 10)
                if type(limit) is not int or not 1 <= limit <= LIMITS['reference_results']:
                    raise ValueError('invalid reference bound')
                result = self.references(args['symbol'], args['path_prefix'], limit)
            raw = canonical(result)
            source_safe(raw.decode())
            if (len(raw) > LIMITS['tool_result_bytes'] or len(raw.decode()) > LIMITS['read_chars'] + 1800 or
                self.returned_chars + len(raw.decode()) > LIMITS['returned_chars'] or
                self.returned_bytes + len(raw) > LIMITS['returned_bytes']):
                return {'error': 'tool_result_or_total_output_cap', 'unknown': True}
            self.returned_chars += len(raw.decode())
            self.returned_bytes += len(raw)
            self.register_evidence(result)
            return result
        except (ValueError, TypeError, KeyError, StopIteration, UnicodeError, SyntaxError, subprocess.SubprocessError):
            return {'error': 'invalid_or_unavailable_source_request', 'unknown': True}

    def register_evidence(self, result):
        if isinstance(result, dict):
            if 'evidence_id' in result:
                self.catalog[result['evidence_id']] = {k: result[k] for k in ('path', 'start_line', 'end_line')}
            for v in result.values():
                self.register_evidence(v)
        elif isinstance(result, list):
            for v in result:
                self.register_evidence(v)


class EventLog:
    def __init__(self, path):
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.last = '0' * 64
        self.n = 0
        if self.path.exists():
            raise ValueError('attempt log exists; use a new attempt directory (never silently retry)')

    def add(self, kind, data):
        event = {'sequence': self.n, 'utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
                 'kind': kind, 'previous_sha256': self.last, 'data': data}
        self.last = sha(event)
        event['event_sha256'] = self.last
        with self.path.open('ab') as f:
            f.write(canonical(event) + b'\n')
            f.flush()
            os.fsync(f.fileno())
        self.n += 1


class SharedBudget:
    """Root-created shared physical ledger. Lock spans each reservation/settlement.

    Unknown usage keeps its full reservation outstanding and poisons further calls.
    External Jev entries must use the same ledger or be reserved/debited by root first.
    """
    def __init__(self, path):
        self.path = Path(path)
        if not self.path.exists():
            raise ValueError('root must create/validate shared budget first')

    def update(self, operation):
        with Path(str(self.path) + '.lock').open('a') as lock:
            fcntl.flock(lock, fcntl.LOCK_EX)
            data = load_json(self.path)
            if data.get('schema') != 1 or not 0 < data['limit_usd'] <= 1 or data.get('halted'):
                raise ValueError('invalid or halted shared physical ledger')
            value = operation(data)
            save(self.path, data)
            return value

    def reserve(self, key, amount):
        def op(data):
            total = sum(e['charged_usd'] if e.get('charged_usd') is not None else e['reserved_usd'] for e in data['entries'])
            if total + amount > data['limit_usd'] + 1e-10:
                return False
            if any(e['key'] == key for e in data['entries']):
                raise ValueError('request attempt already reserved')
            data['entries'].append({'key': key, 'reserved_usd': amount, 'charged_usd': None,
                                    'job': VERSION, 'state': 'reserved'})
            return True
        return self.update(op)

    def settle(self, key, cost):
        def op(data):
            entry = next(e for e in data['entries'] if e['key'] == key)
            entry['charged_usd'] = cost
            entry['state'] = 'usage_unknown' if cost is None else 'settled'
            if cost is None or cost > entry['reserved_usd'] + 1e-10:
                data['halted'] = True
        self.update(op)


def usage_cost(response):
    u = response.get('usage')
    if not isinstance(u, dict) or any(type(u.get(k)) is not int or u[k] < 0 for k in ('input_tokens', 'output_tokens')):
        return None
    # No explicit cache is requested. Unknown cache tariffs stay unknown instead of undercharging.
    if any(u.get(k, 0) for k in ('cache_creation_input_tokens', 'cache_read_input_tokens')):
        return None
    if response.get('model') != MODEL:
        return None
    return round((u['input_tokens'] * 2 + u['output_tokens'] * 10) / 1e6, 8)


def final_decision(response, source):
    if response.get('stop_reason') != 'end_turn':
        return None, 'capped_or_nonfinal_response'
    blocks = response.get('content', [])
    if any(b.get('type') != 'text' for b in blocks):
        return None, 'nontext_final'
    try:
        d = json.loads(''.join(b['text'] for b in blocks))
        required = {'label', 'score', 'owner', 'why', 'uncertainty', 'missing_evidence', 'citations'}
        if not isinstance(d, dict) or set(d) != required:
            raise ValueError()
        if d['label'] not in ('duplicate', 'drifted', 'separate') or d['owner'] not in ('', 'new', 'existing', 'shared'):
            raise ValueError()
        if type(d['score']) not in (int, float) or not math.isfinite(d['score']) or not 0 <= d['score'] <= 1:
            raise ValueError()
        if type(d['uncertainty']) is not bool or not isinstance(d['why'], str) or len(d['why'].split()) > 80:
            raise ValueError()
        if not isinstance(d['missing_evidence'], list) or len(d['missing_evidence']) > 3 or any(not isinstance(s, str) or len(s) > 300 for s in d['missing_evidence']):
            raise ValueError()
        if not isinstance(d['citations'], list) or not 1 <= len(d['citations']) <= 8:
            raise ValueError()
        for cite in d['citations']:
            evidence = source.catalog[cite['evidence_id']]
            if set(cite) != {'evidence_id', 'start_line', 'end_line'} or any(type(cite[k]) is not int for k in ('start_line', 'end_line')):
                raise ValueError()
            if not evidence['start_line'] <= cite['start_line'] <= cite['end_line'] <= evidence['end_line']:
                raise ValueError()
        return d, 'uncertain' if d['uncertainty'] else 'complete'
    except (ValueError, TypeError, KeyError):
        return None, 'malformed_final'


def validate_item(item):
    if item['revision'] != item['source_config']['revision'] or item['repository'] != item['source_config']['repository']:
        raise ValueError('source binding mismatch')
    if sha(item['candidate']) != item['candidate_sha256'] or sha(item['source']) != item['source_identity_sha256']:
        raise ValueError('source/candidate identity mismatch')
    if item['request_identity'] != request_identity(item) or item['request_key'] != sha(request_identity(item)):
        raise ValueError('request or harness changed: requires a new preregistered identity')
    if item['initial_request'] != initial_request(item):
        raise ValueError('rendered request mismatch')
    source = SourceTools(item['source_config'], item['candidate'])
    for proof in item['source']:
        text = source.file(proof['path'])
        if sha(text.encode()) != proof['file_sha256']:
            raise ValueError('pinned file hash mismatch')
        from dry_lint.extractors import PythonExtractor
        from dry_lint.typescript import TypeScriptExtractor
        extractor = PythonExtractor() if proof['path'].endswith('.py') else TypeScriptExtractor()
        f = item['candidate'][proof['side']]
        matched = [g for g in extractor.functions(proof['path'], text)
                   if (g.name, g.owner, g.line, g.end_line) == (f['name'], f.get('owner'), f['line'], f['end_line'])]
        if len(matched) != 1 or matched[0].code != f['code'] or sha(f['code'].encode()) != proof['body_sha256']:
            raise ValueError('recorded function body differs from pinned extraction')
    return source


def run_case(item, out, client, budget):
    started = time.monotonic()
    source = validate_item(item)
    log = EventLog(Path(out) / item['id'] / 'events.jsonl')
    log.add('case_begin', {'id': item['id'], 'request_identity': item['request_identity'],
                           'request_key': item['request_key'], 'initial_request': item['initial_request'],
                           'candidate': item['candidate'], 'source': item['source']})
    request = copy.deepcopy(item['initial_request'])
    total_cost, known_cost, api_seconds, tool_seconds = 0.0, 0.0, 0.0, 0.0
    usages, responses, calls, decision, status = [], [], 0, None, 'round_cap'
    for round_index in range(LIMITS['rounds']):
        size = len(canonical(request))
        reserve = reservation(request)
        if size > LIMITS['context_bytes'] or size + 4096 > LIMITS['context_token_upper_bound']:
            status = 'context_cap'
            break
        if total_cost + reserve > LIMITS['case_usd']:
            status = 'case_budget_deferred'
            break
        round_key = sha({'attempt': str(Path(out).resolve()), 'request_key': item['request_key'],
                         'round': round_index, 'request': request})
        if not budget.reserve(round_key, reserve):
            status = 'shared_budget_deferred'
            break
        log.add('before_provider', {'round': round_index, 'key': round_key, 'request': request,
                                    'request_sha256': sha(request), 'reserved_usd': reserve,
                                    'estimated_usd': estimated_cost(request), 'sdk_retries': 0})
        t = time.monotonic()
        calls += 1
        try:
            response = client.messages.create(**request).model_dump(mode='json')
        except Exception as error:
            elapsed = time.monotonic() - t
            api_seconds += elapsed
            # Never serialize exception text: SDK errors may contain request headers.
            log.add('after_provider_error', {'round': round_index, 'error_type': type(error).__name__,
                                             'seconds': elapsed, 'usage': None, 'cost_usd': None})
            budget.settle(round_key, None)
            total_cost = None
            status = 'provider_error_usage_unknown'
            break
        elapsed = time.monotonic() - t
        cost = usage_cost(response)
        # Durable provider response precedes parsing, tools, and subsequent request.
        log.add('after_provider', {'round': round_index, 'response': response, 'response_sha256': sha(response),
                                   'seconds': elapsed, 'usage': response.get('usage'), 'cost_usd': cost})
        budget.settle(round_key, cost)
        responses.append(sha(response))
        usages.append(response.get('usage'))
        api_seconds += elapsed
        if cost is None:
            status, total_cost = 'usage_or_model_unknown', None
            break
        total_cost += cost
        known_cost += cost
        if cost > reserve:
            status = 'reservation_exceeded'
            break
        if response.get('stop_reason') != 'tool_use':
            decision, status = final_decision(response, source)
            break
        blocks = response.get('content', [])
        requested = [b for b in blocks if b.get('type') == 'tool_use']
        if not requested or len(requested) > LIMITS['tools_per_round'] or len({b.get('id') for b in requested}) != len(requested):
            status = 'malformed_or_capped_tool_round'
            break
        results = []
        tool_failed = False
        for b in requested:
            log.add('before_tool', {'round': round_index, 'tool_use_id': b.get('id'),
                                    'name': b.get('name'), 'input': b.get('input')})
            t = time.monotonic()
            result = source.call(b.get('name'), b.get('input'))
            tool_failed |= bool(result.get('error'))
            duration = time.monotonic() - t
            tool_seconds += duration
            log.add('after_tool', {'round': round_index, 'tool_use_id': b.get('id'), 'result': result,
                                   'result_sha256': sha(result), 'seconds': duration})
            results.append({'type': 'tool_result', 'tool_use_id': b['id'], 'content': canonical(result).decode(),
                            'is_error': bool(result.get('error'))})
        if tool_failed:
            status = 'tool_error_or_cap'
            break
        request['messages'].append({'role': 'assistant', 'content': blocks})
        request['messages'].append({'role': 'user', 'content': results})
    result = {'schema': 1, 'id': item['id'], 'split': item['split'], 'request_key': item['request_key'],
              'status': status, 'decision': decision, 'reported': (decision['label'] in ('duplicate', 'drifted') and
              decision['score'] >= .8) if decision and status == 'complete' else None,
              'cost_usd': total_cost, 'usage': usages, 'provider_calls': calls, 'tool_calls': source.tool_calls,
              'known_cost_usd': round(known_cost, 8),
              'returned_chars': source.returned_chars, 'returned_bytes': source.returned_bytes,
              'api_seconds_sum': api_seconds, 'tool_seconds_sum': tool_seconds,
              'case_wall_seconds': time.monotonic() - started, 'depth': len(usages),
              'response_sha256s': responses, 'evidence_catalog': source.catalog, 'final_event_sha256': log.last,
              'cache': 'fresh/no provider prompt caching requested', 'concurrency': 1}
    result['result_key'] = sha({'request_key': item['request_key'], 'responses': responses,
                              'final_event_sha256': log.last, 'output': decision, 'status': status})
    log.add('case_end', result)
    save(Path(out) / item['id'] / 'result.json', result)
    return result


def validate_prereg(queue):
    prereg = load_json(HERE.parent / 'EXPERIMENT.json')
    if sha(Path(queue).read_bytes()) != prereg['queue_sha256']:
        raise ValueError('queue differs from preregistration')
    if sha((HERE / 'split.json').read_bytes()) != prereg['split_sha256']:
        raise ValueError('split differs from preregistration')
    if sha((HERE / 'references.jsonl').read_bytes()) != prereg['references_sha256']:
        raise ValueError('source references differ from preregistration')
    items = lines(queue)
    if len(items) != 8 or len({i['id'] for i in items}) != 8:
        raise ValueError('exact eight-case preregistration required')
    for item in items:
        validate_item(item)
    return items


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('command', choices=['validate', 'execute', 'init-ledger'])
    p.add_argument('--queue', default=str(HERE / 'queue.jsonl'))
    p.add_argument('--out', default=str(HERE / 'runs' / 'root-attempt-2'))
    p.add_argument('--shared-ledger')
    p.add_argument('--case', action='append', default=[])
    p.add_argument('--limit-usd', type=float, default=1.0)
    p.add_argument('--external-reserved-usd', type=float, default=0.0)
    p.add_argument('--root-reviewed-pricing-and-source', action='store_true')
    args = p.parse_args()
    if args.command == 'init-ledger':
        if not args.shared_ledger or Path(args.shared_ledger).exists() or not 0 < args.limit_usd <= 1 or not 0 <= args.external_reserved_usd <= args.limit_usd:
            p.error('new shared ledger, ceiling <=1 and valid Jev external reservation required')
        entries = [] if not args.external_reserved_usd else [{'key': 'root-reserved-other-job', 'job': 'Jev',
                   'reserved_usd': args.external_reserved_usd, 'charged_usd': None, 'state': 'external_reservation'}]
        save(args.shared_ledger, {'schema': 1, 'limit_usd': args.limit_usd, 'halted': False, 'entries': entries})
        return
    items = validate_prereg(args.queue)
    if args.command == 'validate':
        print(json.dumps({'valid': True, 'cases': len(items), 'provider_calls': 0, 'sdk_initialized': False}))
        return
    if not args.root_reviewed_pricing_and_source or not args.shared_ledger:
        p.error('root execution requires reviewed pricing/source and existing shared physical ledger')
    chosen = [i for i in items if not args.case or i['id'] in args.case]
    if not chosen or set(args.case) - {i['id'] for i in items}:
        p.error('unknown or empty case selection')
    budget = SharedBudget(args.shared_ledger)
    # This branch is for the trusted root coordinator only. Worker never executes it.
    # SDK reads root-owned environment; no key is read, printed, stored or forwarded by tool code.
    import anthropic
    import httpx2
    client = anthropic.Anthropic(max_retries=0, timeout=LIMITS['timeout_seconds'],
                                 base_url='https://api.anthropic.com',
                                 http_client=httpx2.Client(trust_env=False))
    try:
        for item in chosen:
            result = run_case(item, args.out, client, budget)
            print(json.dumps({k: result[k] for k in ('id', 'status', 'provider_calls', 'cost_usd')}), flush=True)
            if result['cost_usd'] is None or result['status'] in ('reservation_exceeded', 'shared_budget_deferred'):
                break
    finally:
        client.close()


if __name__ == '__main__':
    main()
