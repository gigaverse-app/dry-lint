def module(name,path):
    spec=importlib.util.spec_from_file_location(name,path);m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m);return m


def read(p):return json.loads(p.read_text())
def rows(p):return [json.loads(x) for x in p.read_text().splitlines() if x.strip()]
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def freeze(p,data):
    p.parent.mkdir(parents=True,exist_ok=True)
    raw=(json.dumps(data,sort_keys=True,indent=2)+'\n').encode()
    if p.exists():assert p.read_bytes()==raw,('Immutable file changed',str(p))
    else:p.write_bytes(raw)
def lines(p,data):
    p.parent.mkdir(parents=True,exist_ok=True)
    raw=b''.join(q.canonical(r)+b'\n' for r in data)
    if p.exists():assert p.read_bytes()==raw,('Immutable queue changed',str(p))
    else:p.write_bytes(raw)


def pair(repo,pin,ids):return q.sha(q.canonical([repo,pin,sorted(ids)]))
def rendered_key(identity):
    fields=['kind','judge','version','model','max_output_tokens','rendered_input','source_evidence','context_hash']
    return q.sha(q.canonical({k:identity.get(k) for k in fields}))

def group_rows(rr,key):
    result=defaultdict(list)
    for r in rr:result[key(r)].append(r)
    return result
