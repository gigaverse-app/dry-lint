class ProvenanceJudge(h.OfflineJudge):
 def judge(self,candidates,settings):
  start=len(self.events)
  try:return super().judge(candidates,settings)
  finally:
   for event in self.events[start:]:
    row=self.answers.get(event['question_key'],{})
    event['observation_provenance']=row.get('observation_provenance',{'kind':'historical_exact_cache'} if row else {'kind':'unknown'})
